  printJson({
      sessionId: result.session.id,
      state: result.session.state,
      agentRuns: result.specialistRuns.length,
      validatedFindings: result.validatedFindings.length,
      summary: result.session.summary
    });
    return 0;
  } finally {
    await runtime.manager.close();
  }
}
async function resumeCommand(args, cwd) {
  const sessionId = args.find((arg) => !arg.startsWith("-"));
  const runtime = await createRuntime(cwd);
  try {
    const result = await runtime.orchestrator.resume(sessionId);
    printJson({ sessionId: result.session.id, state: result.session.state, validatedFindings: result.validatedFindings.length });
    return 0;
  } finally {
    await runtime.manager.close();
  }
}
async function listStoreCommand(kind, args, cwd) {
  const runtime = await createRuntime(cwd);
  try {
    const sessionId = option(args, "--session");
    const items = kind === "hypotheses" ? await runtime.hypotheses.list(sessionId) : await runtime.findings.list(sessionId);
    args.includes("--json") ? printJson(items) : printTable(items);
    return 0;
  } finally {
    await runtime.manager.close();
  }
}
async function evidenceCommand(args, cwd) {
  const [subcommand = "list", ...rest] = args;
  const runtime = await createRuntime(cwd);
  try {
    if (subcommand === "verify") {
      const result = await runtime.evidence.verify();
      printJson(result);
      return result.valid ? 0 : 1;
    }
    const records = await runtime.evidence.list(option(rest, "--session"));
    const metadata = records.map(({ request: _request, response: _response, ...record }) => record);
    rest.includes("--json") ? printJson(metadata) : printTable(metadata);
    return 0;
  } finally {
    await runtime.manager.close();
  }
}
async function auditCommand(args, cwd) {
  const [subcommand = "verify"] = args;
  if (subcommand !== "verify") throw new Error("Usage: webcat audit verify");
  const runtime = await createRuntime(cwd);
  try {
    const result = await runtime.audit.verify();
    printJson(result);
    return result.valid ? 0 : 1;
  } finally {
    await runtime.manager.close();
  }
}
async function reportCommand(args, cwd) {
  const format = option(args, "--format") ?? "markdown";
  if (!["markdown", "json"].includes(format)) throw new Error("--format must be markdown or json.");
  const runtime = await createRuntime(cwd);
  try {
    const requested = option(args, "--session");
    const session = requested ? await runtime.sessions.get(requested) : await runtime.sessions.latest();
    if (!session) throw new Error("No session exists.");
    const input2 = {
      engagement: runtime.engagement,
      session,
      findings: await runtime.findings.list(session.id),
      hypotheses: await runtime.hypotheses.list(session.id),
      evidence: await runtime.evidence.list(session.id)
    };
    const content = format === "json" ? renderJsonReport(input2) : renderMarkdownReport(input2);
    const output2 = option(args, "--output");
    if (output2) {
      const path = resolve6(cwd, output2);
      await mkdir5(resolve6(path, ".."), { recursive: true });
      await writeFile5(path, content, "utf8");
      console.log(path);
    } else {
      process.stdout.write(content.endsWith("\n") ? content : `${content}
`);
    }
    return 0;
  } finally {
    await runtime.manager.close();
  }
}
function option(args, name) {
  const index = args.indexOf(name);
  return index >= 0 ? args[index + 1] : void 0;
}
