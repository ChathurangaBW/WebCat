function validateEngagement(raw) {

    throw new Error("Engagement schemaVersion must be 1.");
  for (const key of ["id", "name", "authorizedBy", "authorizationReference", "startsAt", "expiresAt", "mode"]) {
    if (typeof raw[key] !== "string" || raw[key].length === 0)
      throw new Error(`Engagement field "${key}" is required.`);
  }
  if (!["observe", "manual", "authorized-auto"].includes(raw.mode))
    throw new Error("Invalid engagement mode.");
  if (!Array.isArray(raw.allow) || raw.allow.length === 0)
    throw new Error("At least one allow scope rule is required.");
  if (!Array.isArray(raw.deny))
    throw new Error("deny must be an array.");

}

function normalizeUrl(value) {
  const candidate = /^[a-z][a-z0-9+.-]*:\/\//i.test(value) ? value : `https://${value}`;
  const url = new URL(candidate);
  url.hostname = url.hostname.toLowerCase().replace(/\.$/, "");
  if (!["http:", "https:", "ws:", "wss:"].includes(url.protocol))
    throw new Error("Unsupported URL scheme.");
  return url;
}
function matches(rule, target, operation) {
  if (!rule.hosts.some((pattern) => hostMatches(pattern, target.hostname)))
    return false;
  if (rule.schemes && !rule.schemes.includes(target.protocol.replace(":", "")))
    return false;
  if (rule.ports && !rule.ports.includes(effectivePort(target)))
    return false;
  if (rule.paths && !rule.paths.some((pattern) => pathMatches(pattern, target.pathname)))
    return false;
  if (rule.methods && operation.method && !rule.methods.map((x) => x.toUpperCase()).includes(operation.method.toUpperCase()))
    return false;
  if (rule.operations && !rule.operations.includes(operation.risk))
    return false;
  return true;
}
function hostMatches(pattern, hostname) {
  const normalizedPattern = pattern.toLowerCase().replace(/\.$/, "");
  const normalizedHost = hostname.toLowerCase().replace(/\.$/, "");
  if (normalizedPattern.startsWith("*.")) {
    const suffix = normalizedPattern.slice(2);
    return normalizedHost.endsWith(`.${suffix}`) && normalizedHost !== suffix;
  }
  return normalizedPattern === normalizedHost;
}
function pathMatches(pattern, pathname) {
  const escaped = pattern.replace(/[.+^${}()|[\]\\]/g, "\\$&").replaceAll("**", "\0").replaceAll("*", "[^/]*").replaceAll("\0", ".*");
  return new RegExp(`^${escaped}$`).test(pathname);
}
function effectivePort(url) {
  if (url.port)

  return url.protocol === "https:" || url.protocol === "wss:" ? 443 : 80;
}
