  if (!["passive", "active", "high", "destructive"].includes(operation)) throw new Error("Invalid --operation.");
  const method = option(args, "--method");
  const external = {
    server: "operator",
    tool: "scope-check",
    capability: "scope.read",
    risk: operation,
    target,
    ...method ? { method } : {},
    arguments: {}
  };
  const decision = new ScopeEngine(await loadEngagement(cwd)).evaluate(external);
  if (args.includes("--json")) printJson(decision);
  else console.log(`${decision.allowed ? "ALLOW" : "DENY"}: ${decision.reason}${decision.normalizedTarget ? `
Target: ${decision.normalizedTarget}` : ""}`);
  return decision.allowed ? 0 : 2;
}

async function mcpCommand(args, cwd) {
  const [subcommand = "status", ...rest] = args;
  const runtime = await createRuntime(cwd);
  try {
    if (subcommand === "status") {
      const statuses = await runtime.manager.status(rest.includes("--connect"));
      rest.includes("--json") ? printJson(statuses) : printTable(statuses);
      return statuses.some((status) => status.error) ? 1 : 0;
    }
    if (subcommand === "tools") {
      const tools = await runtime.manager.listAllTools();
      const rows = tools.map(({ server, tool }) => {
        const descriptor = classifyTool(tool, runtime.manager.getServerConfig(server));
        return { server, tool: tool.name, capability: descriptor.capability, risk: descriptor.risk, description: tool.description ?? "" };
      });
      rest.includes("--json") ? printJson(rows) : printTable(rows);
      return 0;
    }
    if (subcommand === "call") {
      const server = rest[0];
      const tool = rest[1];
      if (!server || !tool) throw new Error("Usage: webcat mcp call <server> <tool> --args '{...}' --session <id>");
      const argsValue = JSON.parse(option(rest, "--args") ?? "{}");
      const sessionId = option(rest, "--session") ?? "operator";
      const result = await runtime.guard.execute({
        server,
        tool,
        arguments: argsValue,
        sessionId,
        actor: "operator",
        reason: option(rest, "--reason") ?? "Explicit operator MCP call."
      });
      printJson(result);
      return 0;
    }
    throw new Error(`Unknown mcp command "${subcommand}".`);
  } finally {
    await runtime.manager.close();
  }
}
async function approvalsCommand(args, cwd) {
  const [subcommand = "list", ...rest] = args;
  const runtime = await createRuntime(cwd);
  try {
    if (subcommand === "list") {
      const approvals = await runtime.approvals.list(rest.includes("--all"));
      rest.includes("--json") ? printJson(approvals) : printTable(approvals);
      return 0;
    }
    if (subcommand === "grant") {
      const risk = option(rest, "--risk");
      const ttl = option(rest, "--ttl") ?? "20m";
      const reason = option(rest, "--reason");
      if (!risk || !["passive", "active", "high", "destructive"].includes(risk) || !reason) {
        throw new Error('Usage: webcat approvals grant --risk active --ttl 20m --reason "..."');
      }
      const capability = option(rest, "--capability");
      const server = option(rest, "--server");
      const tool = option(rest, "--tool");
      const approval = await runtime.approvals.grant({
        risk,
        ttl,
        reason,
        ...capability ? { capability } : {},
        ...server ? { server } : {},
        ...tool ? { tool } : {}
      });
      await runtime.audit.append("approval.granted", "operator", approval);
      printJson(approval);
      return 0;
    }
    if (subcommand === "revoke") {
      const id2 = rest[0];
      if (!id2) throw new Error("Usage: webcat approvals revoke <approval-id>");
      if (!await runtime.approvals.revoke(id2)) throw new Error(`Approval ${id2} not found.`);
      await runtime.audit.append("approval.revoked", "operator", { id: id2 });
      console.log(`Revoked ${id2}.`);
      return 0;
    }
    throw new Error(`Unknown approvals command "${subcommand}".`);
  } finally {
    await runtime.manager.close();
  }
}
