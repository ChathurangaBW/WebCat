  const paths = projectPaths(cwd);
  const [engagement, config, manager] = await Promise.all([
    loadEngagement(cwd),
    loadConfig(cwd),
    McpManager.load(paths.mcp)
  ]);
  const approvals = new ApprovalStore(paths.approvals, engagement.id);
  const audit = new AuditLog(paths.audit);
  const evidence = new EvidenceStore(paths.evidence, config.evidence);
  const sessions = new SessionStore(paths.sessions);
  const hypotheses = new HypothesisStore(paths.hypotheses);
  const findings = new FindingStore(paths.findings);
  const guard = new McpGuard(engagement, manager, approvals, evidence, audit);

    guard,
    provider,
    sessions,
    hypotheses,
    findings,
    evidence,
    audit
  });
  return {
    paths,
    engagement,
    config,
    manager,
    approvals,
    audit,
    evidence,
    sessions,
    hypotheses,
    findings,
    guard,
    provider,
    orchestrator
  };
}
