function tokenize(value) {
  const tokens = [];
  let current = "";
  let quote2;
  let escaped = false;
  for (const char of value) {
    if (escaped) {
      current += char;
      escaped = false;
      continue;
    }
    if (char === "\\") {
      escaped = true;
      continue;
    }
    if (quote2) {
      if (char === quote2) quote2 = void 0;
      else current += char;
      continue;
    }
    if (char === "'" || char === '"') {
      quote2 = char;
      continue;
    }
    if (/\s/.test(char)) {
      if (current) {
        tokens.push(current);
        current = "";
      }
      continue;
    }
    current += char;
  }
  if (quote2) throw new Error("Unterminated quote.");
  if (current) tokens.push(current);
  return tokens;
}

  return JSON.stringify(value);
