      async resume(sessionId) {
        const session = sessionId ? await this.deps.sessions.get(sessionId) : await this.deps.sessions.latest();
        if (!session)
          throw new Error("No session is available to resume.");
        if (session.state === "COMPLETED") {
          return {
            session,
            specialistRuns: [],
            validatedFindings: (await this.deps.findings.list(session.id)).filter((finding) => finding.status === "validated")
          };
        }
        await this.deps.audit.append("session.resume_requested", "operator", { previousState: session.state }, { sessionId: session.id });
        return await this.run(`${session.objective}

Resume context: previous session ${session.id} ended in state ${session.state}.`);
      }
      async authorize(session) {
        try {
          assertAuthorizationCurrent(this.deps.engagement);
        } catch (error) {
          const awaiting = await this.deps.sessions.transition(session, "AUTHORIZATION_REQUIRED");
          await this.deps.sessions.transition(awaiting, "BLOCKED", { error: error.message });
          throw error;
        }
        return await this.deps.sessions.transition(session, "SCOPE_READY");
      }
