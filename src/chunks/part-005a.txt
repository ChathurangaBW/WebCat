  constructor(responder) {
        this.responder = responder;
      }
      async complete(messages) {
        if (this.responder)
          return await this.responder(messages);
        const profile2 = /Profile:\s*([^\n]+)/.exec(messages[0]?.content ?? "")?.[1] ?? "mock";
        return JSON.stringify({
          type: "result",
          handoff: {
            profile: profile2,
            summary: "Mock provider completed without external tool calls.",
            hypotheses: [],
            candidateFindings: [],
            notes: ["Mock provider was used; no security conclusion was produced."]
          }
        });
      }
    };
  }
});

function parseAgentAction(text, profile2) {
  const parsed = parseJson(text);
  if (!isRecord(parsed))
    throw new Error("Agent response must be a JSON object.");
  if (parsed.type === "tool") {
    if (typeof parsed.server !== "string" || typeof parsed.tool !== "string" || !isRecord(parsed.arguments)) {
      throw new Error("Tool action requires server, tool, and object arguments.");
    }
    return {
      type: "tool",
      server: parsed.server,
      tool: parsed.tool,
      arguments: parsed.arguments,
      reason: typeof parsed.reason === "string" ? parsed.reason : "No reason supplied."
    };
  }
  if (parsed.type !== "result" || !isRecord(parsed.handoff))
    throw new Error("Agent response must be a tool action or result handoff.");
  return { type: "result", handoff: normalizeHandoff(parsed.handoff, profile2) };
}
function parseJson(text) {
  const trimmed = text.trim();
  try {
    return JSON.parse(trimmed);
  } catch {
    const fenced = /```(?:json)?\s*([\s\S]*?)```/i.exec(trimmed)?.[1];
    if (fenced)
      return JSON.parse(fenced);
    const start = trimmed.indexOf("{");
    const end = trimmed.lastIndexOf("}");
    if (start >= 0 && end > start)
      return JSON.parse(trimmed.slice(start, end + 1));
    throw new Error("Agent response did not contain valid JSON.");
  }
}
function normalizeHandoff(raw, profile2) {
  return {
    profile: profile2,
    summary: typeof raw.summary === "string" ? raw.summary : "No summary supplied.",
    hypotheses: Array.isArray(raw.hypotheses) ? raw.hypotheses.map((item) => normalizeHypothesis(item, profile2)).filter(Boolean) : [],
    candidateFindings: Array.isArray(raw.candidateFindings) ? raw.candidateFindings.map(normalizeFinding).filter(Boolean) : [],
    notes: Array.isArray(raw.notes) ? raw.notes.filter((item) => typeof item === "string") : []
  };
}
function normalizeHypothesis(value, profile2) {
  if (!isRecord(value) || typeof value.title !== "string" || typeof value.rationale !== "string")
    return void 0;
  return {
    title: value.title,
    rationale: value.rationale,
    profile: profile2,
    status: ["candidate", "testing", "supported", "rejected"].includes(String(value.status)) ? value.status : "candidate",
    ...typeof value.target === "string" ? { target: value.target } : {},
    evidenceIds: Array.isArray(value.evidenceIds) ? value.evidenceIds.filter((item) => typeof item === "string") : []
  };
}
function normalizeFinding(value) {
  if (!isRecord(value) || typeof value.title !== "string" || typeof value.description !== "string")
    return void 0;
  const severity = ["info", "low", "medium", "high", "critical"].includes(String(value.severity)) ? value.severity : "info";
  const status = ["candidate", "validated", "rejected", "accepted-risk"].includes(String(value.status)) ? value.status : "candidate";
  return {
    title: value.title,
    severity,
    status,
    category: typeof value.category === "string" ? value.category : "Uncategorized",
    ...typeof value.target === "string" ? { target: value.target } : {},
    description: value.description,
    impact: typeof value.impact === "string" ? value.impact : "Impact not established.",
    remediation: typeof value.remediation === "string" ? value.remediation : "Review and address the underlying control.",
    reproduction: Array.isArray(value.reproduction) ? value.reproduction.filter((item) => typeof item === "string") : [],
    evidenceIds: Array.isArray(value.evidenceIds) ? value.evidenceIds.filter((item) => typeof item === "string") : [],
    confidence: clamp(typeof value.confidence === "number" ? value.confidence : 0.25, 0, 1),
    ...typeof value.validatorNotes === "string" ? { validatorNotes: value.validatorNotes } : {},
    ...typeof value.criticNotes === "string" ? { criticNotes: value.criticNotes } : {}
  };
}
