function summarize(operation) {
  const { arguments: args, ...rest } = operation;
  return { ...rest, argumentKeys: Object.keys(args) };
}
function filterOutOfScope(value, scope) {
  let removed = 0;
  const filtered = visit(value, (candidate) => {
    const target = candidateTarget(candidate);
    if (target && !scope.isTargetAllowed(target)) {
      removed += 1;
      return void 0;
    }
    return candidate;
  });
  return { value: filtered ?? { content: [], isError: false }, removed };
}
function visit(value, filter) {
  if (Array.isArray(value)) {
    return value.map((item) => visit(item, filter)).filter((item) => item !== void 0);
  }
  if (value && typeof value === "object") {
    const record = value;
    const filtered = filter(record);
    if (!filtered)
      return void 0;
    return Object.fromEntries(Object.entries(filtered).map(([key, item]) => [key, visit(item, filter)]).filter(([, item]) => item !== void 0));
  }
  return value;
}
function candidateTarget(record) {
  for (const key of ["url", "uri", "target", "requestUrl", "endpoint"]) {
    if (typeof record[key] === "string" && /^(?:https?|wss?):\/\//i.test(record[key]))
      return record[key];
  }
  if (typeof record.host === "string") {
    const path = typeof record.path === "string" ? record.path : "/";
    return `https://${record.host}${path.startsWith("/") ? path : `/${path}`}`;
  }
  return void 0;
}
