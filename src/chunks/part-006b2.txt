n await skillsCommand(rest, cwd);
    case "mcp":

    case "approvals":

    case "run":
      return await runCommand(rest, cwd);
    case "resume":

    case "hypotheses":

    default:

  }

async function initCommand(args, cwd) {
  const written = await initializeProject(cwd, args.includes("--force"));
  if (written.length === 0) console.log("WebCat project already initialized. Use --force to replace templates.");
  else {
    console.log("Initialized WebCat:");

async function doctorCommand(args, cwd) {
  const checks = [];
  const paths = projectPaths(cwd);
  let engagement;
  let config;
  try {
    engagement = await loadEngagement(cwd);
    checks.push({ check: "engagement", ok: true, detail: paths.engagement });
    try {
      assertAuthorizationCurrent(engagement);
      checks.push({ check: "authorization-window", ok: true, detail: `${engagement.startsAt} \u2014 ${engagement.expiresAt}` });
    } catch (error) {
      checks.push({ check: "authorization-window", ok: false, detail: error.message });
    }
  } catch (error) {
    checks.push({ check: "engagement", ok: false, detail: error.message });
  }
  try {
    config = await loadConfig(cwd);
    checks.push({ check: "config", ok: true, detail: paths.config });
    const apiKeyReady = config.model.provider === "mock" || Boolean(process.env[config.model.apiKeyEnv]);
    checks.push({ check: "model-credentials", ok: apiKeyReady, detail: config.model.provider === "mock" ? "Mock provider" : config.model.apiKeyEnv });
  } catch (error) {
    checks.push({ check: "config", ok: false, detail: error.message });
  }
  try {
    const manager = await McpManager.load(paths.mcp);
    checks.push({ check: "mcp-config", ok: true, detail: `${Object.keys(manager.config.servers).length} server(s)` });
  } catch (error) {
    checks.push({ check: "mcp-config", ok: false, detail: error.message });
  }
  try {
    const runtime = engagement && config ? await createRuntime(cwd) : void 0;
    const audit = runtime ? await runtime.audit.verify() : { valid: true, records: 0 };
    checks.push({ check: "audit-chain", ok: audit.valid, detail: `${audit.records} record(s)${audit.error ? `: ${audit.error}` : ""}` });
    await runtime?.manager.close();
  } catch (error) {
    checks.push({ check: "runtime", ok: false, detail: error.message });
  }
  if (args.includes("--json")) printJson(checks);
  else printTable(checks);
  return checks.every((check) => check.ok) ? 0 : 1;
}
