function redact(value, config, key = "") {
  const normalizedKey = key.toLowerCase();
  if (config.redactKeys.some((candidate) => normalizedKey === candidate.toLowerCase()) || config.redactHeaders.some((candidate) => normalizedKey === candidate.toLowerCase())) {
    return REDACTED;
  }
  if (typeof value === "string") {
    return value.replace(/(authorization:\s*)([^\r\n]+)/gi, `$1${REDACTED}`).replace(/(cookie:\s*)([^\r\n]+)/gi, `$1${REDACTED}`).replace(/(set-cookie:\s*)([^\r\n]+)/gi, `$1${REDACTED}`).replace(/((?:api[_-]?key|access[_-]?token|refresh[_-]?token|password)=)([^&\s]+)/gi, `$1${REDACTED}`);
  }
  if (Array.isArray(value))
    return value.map((item) => redact(item, config));
  if (isRecord(value)) {
    return Object.fromEntries(Object.entries(value).map(([entryKey, item]) => [entryKey, redact(item, config, entryKey)]));
  }
  return value;
}
function truncate(value, maxBytes) {
  const serialized = JSON.stringify(value);
  if (Buffer.byteLength(serialized) <= maxBytes)
    return { value, truncated: false };
  const responseText = JSON.stringify(value.response);
  const budget = Math.max(256, maxBytes - Buffer.byteLength(JSON.stringify(value.request ?? null)) - 128);
  return {
    value: {
      ...value.request !== void 0 ? { request: value.request } : {},
      response: `${responseText.slice(0, budget)}\u2026[TRUNCATED]`
    },
    truncated: true
  };
}

async function scan(directory) {
  try {
    const entries = await readdir3(directory, { withFileTypes: true });
    const skills = [];
    for (const entry of entries) {
      const candidate = entry.isDirectory() ? resolve4(directory, entry.name, "SKILL.md") : resolve4(directory, entry.name);
      if (!candidate.endsWith(".md"))
        continue;
      try {
        const content = await readFile6(candidate, "utf8");
        const frontmatter = /^---\s*\n([\s\S]*?)\n---\s*\n/.exec(content)?.[1] ?? "";
        const name = /^name:\s*(.+)$/m.exec(frontmatter)?.[1]?.trim() ?? basename(candidate, ".md");
        const description = /^description:\s*(.+)$/m.exec(frontmatter)?.[1]?.trim() ?? firstParagraph(content);
        skills.push({ name, description, path: candidate, content });
      } catch {
      }
    }
    return skills;
  } catch (error) {
    if (error.code === "ENOENT")
      return [];
    throw error;
  }
}
